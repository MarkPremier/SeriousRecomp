cmake_minimum_required(VERSION 3.20)
project(serious_sam_manual_tests C)
set(XBOXRECOMP_DIR "${CMAKE_CURRENT_LIST_DIR}/../../xboxrecomp_ss" CACHE PATH "Runtime headers")
add_executable(fence_submit_test fence_submit_test.c ../src/recomp_manual.c)
add_executable(stream_close_test stream_close_test.c ../src/recomp_manual.c)
add_executable(dsound_idle_test dsound_idle_test.c ../src/recomp_manual.c)
add_executable(menu_present_test menu_present_test.c ../src/recomp_manual.c)
foreach(test_target fence_submit_test stream_close_test dsound_idle_test menu_present_test)
target_include_directories(${test_target} PRIVATE "${XBOXRECOMP_DIR}/include" "${XBOXRECOMP_DIR}/src")
foreach(module kernel d3d audio input apu nv2a)
    target_include_directories(${test_target} PRIVATE "${XBOXRECOMP_DIR}/src/${module}")
endforeach()
endforeach()
enable_testing()
add_test(NAME fence_submit COMMAND fence_submit_test)
add_test(NAME stream_close COMMAND stream_close_test)
add_test(NAME dsound_idle COMMAND dsound_idle_test)
add_test(NAME menu_present COMMAND menu_present_test)
