# Exercise the real runtime with synthetic PCM and D3D11 WARP, without saves
# or game data. Build the Release runtime first for its resampler library.
set(SAM_RUNTIME_BUILD_DIR "${CMAKE_CURRENT_LIST_DIR}/../build/xboxrecomp" CACHE PATH "Built runtime libraries")
add_executable(apu_voices_test apu_voices_test.c)
add_executable(gpu_shader_cache_test gpu_shader_cache_test.cpp)
add_executable(game_files_test game_files_test.c)
foreach(test_target apu_voices_test gpu_shader_cache_test game_files_test)
  target_include_directories(${test_target} PRIVATE "${XBOXRECOMP_DIR}/src")
  target_compile_definitions(${test_target} PRIVATE _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    target_compile_options(${test_target} PRIVATE /Gy /UNDEBUG)
    target_link_options(${test_target} PRIVATE /OPT:REF)
  endif()
endforeach()
target_compile_features(gpu_shader_cache_test PRIVATE cxx_std_17)
target_link_libraries(gpu_shader_cache_test PRIVATE d3d11 dxgi d3dcompiler user32)
target_link_libraries(apu_voices_test PRIVATE "${SAM_RUNTIME_BUILD_DIR}/src/apu/$<CONFIG>/xbox_apu.lib")
target_link_libraries(game_files_test PRIVATE user32 gdi32 comdlg32 bcrypt)
add_test(NAME apu_voices COMMAND apu_voices_test)
add_test(NAME gpu_shader_cache COMMAND gpu_shader_cache_test)
add_test(NAME game_files COMMAND game_files_test)
